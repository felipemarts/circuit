# Static multi-page site: unknown paths are real 404s (no SPA fallback, which made every
# URL answer 200 with the landing page). Share links live in the URL fragment, so the
# server never needs to route them.
ErrorDocument 404 /404.html
Options -Indexes

<IfModule mod_headers.c>
  # Pages change on every deploy; hashed assets under /assets/ are cached by nginx.
  <FilesMatch "\.html$">
    Header set Cache-Control "no-cache"
  </FilesMatch>
</IfModule>
